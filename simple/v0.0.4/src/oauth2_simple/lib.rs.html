16/// Config struct for `Oauth2Simple::with_cfg()`. Using `Cfg::builder()` allows clients to start
17/// with a default config and just override the parts they want (`scope` is required, though).
18#[derive(Builder)]
19pub struct Cfg<'a> {
20    /// Path to a "client info" JSON file.
21    #[builder(default = "./client_info.json")]
22    pub client_info_path: &'a str,
23
24    /// Path to a tokens JSON file.
25    #[builder(default = "./token.json")]
26    pub tokens_path: &'a str,
27
28    /// Port for the http server to listen on.
29    #[builder(default = 42684)]
30    pub port: u16,
31
32    /// OAuth2 API scope to authorize.
33    pub scope: &'a str,
34}

36/// Main struct for this library. Wraps an `oauth2::Oauth2`.
37pub struct Oauth2Simple {
38    /// The underlying `oauth2::Oauth2` struct with the actual logic.
39    oauth2: Oauth2,
40
41    /// Path to a tokens JSON file.
42    tokens_path: String,
43
44    /// Access and refresh token data.
45    tokens: Option<Tokens>,
46}
47
48impl Oauth2Simple {
49    /// Create a new instance. The `tokens_path` can point at a file that doesn't exist yet - one
50    /// will be created when calling `init()`. `refresh()` should only be called if `tokens_path`
51    /// already exists or after `init()` is called.
52    pub fn new(client_info_path: &Path, tokens_path: &Path, scope: &str) -> anyhow::Result<Self> {
53        let cfg = Cfg::builder()
54            .client_info_path(
55                client_info_path
56                    .to_str()
57                    .ok_or(anyhow::anyhow!("client_info_path is invalid"))?,
58            )
59            .tokens_path(
60                tokens_path
61                    .to_str()
62                    .ok_or(anyhow::anyhow!("tokens_path is invalid"))?,
63            )
64            .scope(scope)
65            .build();
66        Self::with_cfg(cfg)
67    }
68
69    /// Make a new instance with the given `Cfg`.
70    pub fn with_cfg(cfg: Cfg) -> anyhow::Result<Self> {
71        let client_info = read_to_string(cfg.client_info_path)?;
72        let client_info: ClientInfo = serde_json::from_str(&client_info)?;
73
74        // `cfg.tokens_path` may or may not exist yet. If it does, use its values, otherwise use
75        // `None`.
76        let (refresh_token, tokens) = if let Ok(e) = exists(cfg.tokens_path)
77            && e
78        {
79            let tokens = read_to_string(cfg.tokens_path)?;
80            let tokens: Tokens = serde_json::from_str(&tokens)?;
81            (Some(tokens.refresh_token.clone()), Some(tokens))
82        } else {
83            (None, None)
84        };
85
86        Ok(Self {
87            oauth2: Oauth2::new(oauth2::Cfg {
88                listen_port: Some(cfg.port),
89                client_id: client_info.installed.client_id,
90                scope: Some(cfg.scope.to_string()),
91                auth_server_url: Some(client_info.installed.auth_uri),
92                token_server_url: client_info.installed.token_uri,
93                client_secret: client_info.installed.client_secret,
94                refresh_token,
95            }),
96            tokens_path: cfg.tokens_path.to_string(),
97            tokens,
98        })
99    }
100
101    /// Performs the initialization flow to create new `oauth2::Tokens` data from an
102    /// `oauth2::ClientInfo`. Results are returned and also written to `tokens_path`.
103    pub async fn init(&mut self) -> anyhow::Result<Tokens> {
104        // Generate a URL that we can show the user, and start up an HTTP server.
105        let (url, http_server_future) = self.oauth2.auth_get_url_and_listen()?;
106        eprintln!("Open {} in your browser.", url);
107        eprintln!();
108
109        // Block until the user clicks on the link and goes through the authorization flow in their
110        // browser. Once they do, their browser will redirect to our local HTTP server's URL, which
111        // will resolve the `Future`.
112        let http_server_result = http_server_future.await?;
113
114        // Exchange `http_server_result` for an access token and a refresh token.
115        let exchange_response = self.oauth2.auth_exchange(http_server_result).await?;
116        let token_result = self
117            .oauth2
118            .auth_handle_exchange_response(exchange_response)
119            .await?;
120
121        // Update our internal state with the new tokens.
122        self.tokens = Some(token_result.clone());
123
124        // Write the result out to `tokens_path`.
125        self.write_tokens(&token_result)?;
126
127        Ok(token_result)
128    }
129
130    /// Performs the refresh flow to get new `Tokens` data from a refresh token. Must only be
131    /// called if the `tokens_path` value in the `Cfg` pointed to a pre-existing file at the time
132    /// of the `Oauth2Simple` construction, or if `init()` has already been called on this
133    /// instance. Results are returned and also written to `tokens_path`.
134    pub async fn refresh(&mut self) -> anyhow::Result<Tokens> {
135        // Get a new access token from a refresh token.
136        let resp = self.oauth2.refresh().await?;
137        let new_token_result = self.oauth2.refresh_handle_response(resp).await?;
138
139        // Write the new data back into `self.tokens`
140        let tokens = self
141            .tokens
142            .as_mut()
143            .ok_or(anyhow::anyhow!(TOKENS_UNSET_ERROR_STR))?;
144        tokens.merge(&new_token_result);
145
146        // Have to redo this to get an immutable ref, otherwise we can't call `write_tokens()`.
147        // Annoying, but that's Rust for you.
148        let tokens = self
149            .tokens
150            .as_ref()
151            .ok_or(anyhow::anyhow!(TOKENS_UNSET_ERROR_STR))?;
152
153        // Write the result out to `tokens_path`.
154        self.write_tokens(tokens)?;
155
156        Ok(tokens.clone())
157    }
158
159    /// Performs the initialization flow as in `init()`, but only if the file at `tokens_path`
160    /// existed upon `Oauth2Simple` initialization.
161    pub async fn init_if_needed(&mut self) -> anyhow::Result<Tokens> {
162        if let Some(tokens) = self.tokens.as_ref() {
163            Ok(tokens.clone())
164        } else {
165            self.init().await
166        }
167    }
168
169    /// Refreshes the access token as in `refresh()`, but only if the previous access token has
170    /// expired or is getting close to expiring. Fails if there was no previous access token.
171    pub async fn refresh_if_needed(&mut self) -> anyhow::Result<Tokens> {
172        let tokens = self
173            .tokens
174            .as_ref()
175            .ok_or(anyhow::anyhow!(TOKENS_UNSET_ERROR_STR))?;
176        if tokens.expires_at > chrono::Local::now() - Duration::from_mins(5) {
177            Ok(tokens.clone())
178        } else {
179            // TODO: handle the case of the refresh token having already expired.
180            self.refresh().await
181        }
182    }
183
184    /// Gets an access token. If there was no existing access token, it does this by running the
185    /// `init()` flow. If the previous access token expired already, then it instead runs the
186    /// `refresh()` flow. Otherwise, it just returns the existing not-yet-expired token.
187    pub async fn get_token(&mut self) -> anyhow::Result<Tokens> {
188        self.init_if_needed().await?;
189        self.refresh_if_needed().await
190    }
191
192    /// Write a `Tokens` struct out to JSON with correct permissions (on *nix systems).
193    fn write_tokens(&self, tokens: &Tokens) -> anyhow::Result<()> {
194        let js = serde_json::to_string_pretty(&tokens)?;
195
196        let mut f = File::create(&self.tokens_path)?;
197
198        #[cfg(unix)]
199        {
200            let md = f.metadata()?;
201            let mut perms = md.permissions();
202            perms.set_mode(0o600);
203            f.set_permissions(perms)?;
204        }
205
206        f.write_all(&js.into_bytes())?;
207
208        Ok(())
209    }
